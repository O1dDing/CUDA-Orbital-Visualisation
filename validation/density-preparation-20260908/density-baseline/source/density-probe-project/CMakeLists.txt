cmake_minimum_required(VERSION 3.24)
cmake_policy(SET CMP0091 NEW)
project(cov_density_source_probe LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT DEFINED COV_PRODUCTION_INCLUDE OR NOT DEFINED COV_PRODUCTION_LIBRARY OR NOT DEFINED COV_CUDART_LIBRARY)
    message(FATAL_ERROR "Use explicit archived production headers and libraries")
endif()
add_executable(cov_density_source_probe ../density_source_probe.cpp)
target_include_directories(cov_density_source_probe PRIVATE "${COV_PRODUCTION_INCLUDE}")
target_compile_features(cov_density_source_probe PRIVATE cxx_std_20)
target_compile_options(cov_density_source_probe PRIVATE /W4 /permissive- /utf-8)
target_link_libraries(cov_density_source_probe PRIVATE "${COV_PRODUCTION_LIBRARY}" "${COV_CUDART_LIBRARY}")
